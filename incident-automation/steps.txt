groups:
  - name: devops-agent-alerts

    rules:
      - alert: HighCPU

        expr: |
          100 - (
            avg by(instance) (
              rate(node_cpu_seconds_total{mode="idle"}[1m])
            ) * 100
          ) > 20

        for: 30s

        labels:
          severity: critical

        annotations:
          summary: "High CPU detected"
          description: "CPU utilization exceeded 20%"
